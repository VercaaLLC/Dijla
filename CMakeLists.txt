cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(Dijla VERSION 2.0.0 LANGUAGES CXX RC)

if(NOT MSVC)
  message(FATAL_ERROR "The portable release requires Microsoft C++ and its static CRT.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with -A Win32 for the Windows 7 through 11 x86 release.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /EHsc /GS)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)

add_library(dijla_core STATIC src/data.cpp)
target_include_directories(dijla_core PUBLIC src)
target_link_libraries(dijla_core PUBLIC bcrypt kernel32)
add_executable(Dijla WIN32 src/ui.cpp src/app.rc)
target_link_libraries(Dijla PRIVATE dijla_core user32 gdi32 comctl32 comdlg32 shell32)
target_link_options(Dijla PRIVATE /SUBSYSTEM:WINDOWS,6.01 /DYNAMICBASE /NXCOMPAT)

add_executable(DijlaTests tests/core_tests.cpp)
target_link_libraries(DijlaTests PRIVATE dijla_core)
target_link_options(DijlaTests PRIVATE /SUBSYSTEM:CONSOLE,6.01 /DYNAMICBASE /NXCOMPAT)
enable_testing()
add_test(NAME core COMMAND DijlaTests "${CMAKE_BINARY_DIR}/test-work" "$<TARGET_FILE:Dijla>")
set_tests_properties(core PROPERTIES TIMEOUT 120)
